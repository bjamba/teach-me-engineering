1 — Word search II (trie + DFS)

Given a grid of letters and a list of words, return which words appear in the grid (4-directionally connected). Brute force is O(words × grid × wordLen). Building a trie of the words collapses this to O(grid × maxWordLen) — pruning DFS branches that no word starts with.

def find_words(board, words):
    # Build trie of the words.
    trie = {}
    for w in words:
        node = trie
        for ch in w:
            node = node.setdefault(ch, {})
        node['$'] = w   # store the full word at the terminal

    rows, cols = len(board), len(board[0])
    found = set()

    def dfs(r, c, node):
        ch = board[r][c]
        if ch not in node: return
        next_node = node[ch]
        if '$' in next_node:
            found.add(next_node['$'])
            # don't return — there may be longer words sharing this path
        board[r][c] = '#'   # mark visited
        for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < rows and 0 <= nc < cols and board[nr][nc] != '#':
                dfs(nr, nc, next_node)
        board[r][c] = ch    # restore

    for r in range(rows):
        for c in range(cols):
            dfs(r, c, trie)

    return sorted(found)
  
function findWords(board, words) {
  const trie = {};
  for (const w of words) {
    let n = trie;
    for (const ch of w) { n[ch] = n[ch] || {}; n = n[ch]; }
    n.$ = w;
  }
  const rows = board.length, cols = board[0].length;
  const found = new Set();

  function dfs(r, c, node) {
    const ch = board[r][c];
    const next = node[ch];
    if (!next) return;
    if (next.$) found.add(next.$);
    board[r][c] = '#';
    for (const [dr, dc] of [[-1,0],[1,0],[0,-1],[0,1]]) {
      const nr = r + dr, nc = c + dc;
      if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && board[nr][nc] !== '#') {
        dfs(nr, nc, next);
      }
    }
    board[r][c] = ch;
  }
  for (let r = 0; r < rows; r++)
    for (let c = 0; c < cols; c++) dfs(r, c, trie);

  return [...found].sort();
}
  
function findWords(board: string[][], words: string[]): string[] {
  type Node = { [k: string]: Node | string | undefined };
  const trie: Node = {};
  for (const w of words) {
    let n = trie;
    for (const ch of w) {
      if (!n[ch] || typeof n[ch] === 'string') n[ch] = {};
      n = n[ch] as Node;
    }
    (n as any).$ = w;
  }
  const rows = board.length, cols = board[0].length;
  const found = new Set<string>();

  function dfs(r: number, c: number, node: Node): void {
    const ch = board[r][c];
    const next = node[ch];
    if (!next || typeof next === 'string') return;
    const word = (next as any).$;
    if (typeof word === 'string') found.add(word);
    board[r][c] = '#';
    for (const [dr, dc] of [[-1,0],[1,0],[0,-1],[0,1]] as [number, number][]) {
      const nr = r + dr, nc = c + dc;
      if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && board[nr][nc] !== '#') {
        dfs(nr, nc, next as Node);
      }
    }
    board[r][c] = ch;
  }
  for (let r = 0; r < rows; r++)
    for (let c = 0; c < cols; c++) dfs(r, c, trie);

  return [...found].sort();
}
O(rows × cols × 4^L) worst case where L = longest word — the trie keeps the constant factor very low because dead branches prune instantly. The temporary '#' marker is a classic in-place visited-tracking trick. Sorting the result is just to make the output deterministic for tests.

2 — Account merge (Union-Find)

A list of accounts, each with a name and emails. Two accounts belong to the same person if they share any email. Merge them. Return the unified account list.

def accounts_merge(accounts):
    parent = {}
    name_of = {}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb: parent[rb] = ra

    # Initialize: each email points to itself; record name.
    for acc in accounts:
        name = acc[0]
        first = acc[1]
        for email in acc[1:]:
            if email not in parent:
                parent[email] = email
            name_of[email] = name
            union(first, email)

    # Group emails by root.
    groups = {}
    for email in parent:
        root = find(email)
        groups.setdefault(root, []).append(email)

    return sorted(
        [[name_of[root]] + sorted(emails) for root, emails in groups.items()],
        key=lambda x: (x[0], x[1])
    )
  
function accountsMerge(accounts) {
  const parent = new Map();
  const nameOf = new Map();

  function find(x) {
    while (parent.get(x) !== x) { parent.set(x, parent.get(parent.get(x))); x = parent.get(x); }
    return x;
  }
  function union(a, b) {
    const ra = find(a), rb = find(b);
    if (ra !== rb) parent.set(rb, ra);
  }

  for (const acc of accounts) {
    const name = acc[0], first = acc[1];
    for (let i = 1; i < acc.length; i++) {
      const email = acc[i];
      if (!parent.has(email)) parent.set(email, email);
      nameOf.set(email, name);
      union(first, email);
    }
  }

  const groups = new Map();
  for (const email of parent.keys()) {
    const root = find(email);
    if (!groups.has(root)) groups.set(root, []);
    groups.get(root).push(email);
  }

  const out = [];
  for (const [root, emails] of groups) {
    emails.sort();
    out.push([nameOf.get(root), ...emails]);
  }
  out.sort((a, b) => a[0] === b[0] ? (a[1] < b[1] ? -1 : 1) : (a[0] < b[0] ? -1 : 1));
  return out;
}
  
function accountsMerge(accounts: string[][]): string[][] {
  const parent = new Map<string, string>();
  const nameOf = new Map<string, string>();

  function find(x: string): string {
    while (parent.get(x) !== x) { parent.set(x, parent.get(parent.get(x)!)!); x = parent.get(x)!; }
    return x;
  }
  function union(a: string, b: string): void {
    const ra = find(a), rb = find(b);
    if (ra !== rb) parent.set(rb, ra);
  }

  for (const acc of accounts) {
    const name = acc[0], first = acc[1];
    for (let i = 1; i < acc.length; i++) {
      const email = acc[i];
      if (!parent.has(email)) parent.set(email, email);
      nameOf.set(email, name);
      union(first, email);
    }
  }

  const groups = new Map<string, string[]>();
  for (const email of parent.keys()) {
    const root = find(email);
    if (!groups.has(root)) groups.set(root, []);
    groups.get(root)!.push(email);
  }

  const out: string[][] = [];
  for (const [root, emails] of groups) {
    emails.sort();
    out.push([nameOf.get(root)!, ...emails]);
  }
  out.sort((a, b) => a[0] === b[0] ? (a[1] < b[1] ? -1 : 1) : (a[0] < b[0] ? -1 : 1));
  return out;
}
O(N × α(N)) where N is total emails — basically O(N). DSU keys are the email strings themselves. The trick is that within each account, every email is unioned with the first one; that automatically chains transitive merges across accounts. Sorting at the end is for deterministic output.

3 — Number of provinces (Union-Find or DFS)

Given an n×n adjacency matrix where isConnected[i][j] = 1 iff city i and city j are directly connected, return the number of provinces (connected components).

def find_circle_num(is_connected):
    n = len(is_connected)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra == rb: return False
        parent[rb] = ra
        return True

    components = n
    for i in range(n):
        for j in range(i + 1, n):
            if is_connected[i][j] and union(i, j):
                components -= 1
    return components
  
function findCircleNum(isConnected) {
  const n = isConnected.length;
  const parent = Array.from({ length: n }, (_, i) => i);
  function find(x) { while (parent[x] !== x) { parent[x] = parent[parent[x]]; x = parent[x]; } return x; }
  let components = n;
  for (let i = 0; i < n; i++) {
    for (let j = i + 1; j < n; j++) {
      if (isConnected[i][j]) {
        const ra = find(i), rb = find(j);
        if (ra !== rb) { parent[rb] = ra; components--; }
      }
    }
  }
  return components;
}
  
function findCircleNum(isConnected: number[][]): number {
  const n = isConnected.length;
  const parent: number[] = Array.from({ length: n }, (_, i) => i);
  function find(x: number): number {
    while (parent[x] !== x) { parent[x] = parent[parent[x]]; x = parent[x]; }
    return x;
  }
  let components = n;
  for (let i = 0; i < n; i++) {
    for (let j = i + 1; j < n; j++) {
      if (isConnected[i][j]) {
        const ra = find(i), rb = find(j);
        if (ra !== rb) { parent[rb] = ra; components--; }
      }
    }
  }
  return components;
}
O(n² × α(n)) ≈ O(n²) time. Could also be solved with DFS/BFS — DSU is here because the same code generalizes to the "edges arrive over time" version (online algorithm), where DFS would have to re-traverse from scratch each time.

4 — LRU cache (full LeetCode 146)

Same as Lesson 3, but here run the LeetCode test suite. Confirm O(1) get and put.

from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity):
        self.cap = capacity
        self.d = OrderedDict()

    def get(self, key):
        if key not in self.d: return -1
        self.d.move_to_end(key)
        return self.d[key]

    def put(self, key, value):
        if key in self.d:
            self.d.move_to_end(key)
        elif len(self.d) >= self.cap:
            self.d.popitem(last=False)
        self.d[key] = value

def run(capacity, ops):
    c = LRUCache(capacity)
    out = []
    for op in ops:
        if op[0] == 'put': c.put(op[1], op[2])
        elif op[0] == 'get': out.append(c.get(op[1]))
    return out
  
class LRUCache {
  constructor(capacity) { this.cap = capacity; this.m = new Map(); }
  get(key) {
    if (!this.m.has(key)) return -1;
    const v = this.m.get(key);
    this.m.delete(key); this.m.set(key, v);
    return v;
  }
  put(key, value) {
    if (this.m.has(key)) this.m.delete(key);
    else if (this.m.size >= this.cap) this.m.delete(this.m.keys().next().value);
    this.m.set(key, value);
  }
}
function run(capacity, ops) {
  const c = new LRUCache(capacity);
  const out = [];
  for (const op of ops) {
    if (op[0] === 'put') c.put(op[1], op[2]);
    else if (op[0] === 'get') out.push(c.get(op[1]));
  }
  return out;
}
  
class LRUCache {
  private m = new Map<number, number>();
  constructor(private cap: number) {}
  get(key: number): number {
    if (!this.m.has(key)) return -1;
    const v = this.m.get(key)!;
    this.m.delete(key); this.m.set(key, v);
    return v;
  }
  put(key: number, value: number): void {
    if (this.m.has(key)) this.m.delete(key);
    else if (this.m.size >= this.cap) this.m.delete(this.m.keys().next().value as number);
    this.m.set(key, value);
  }
}
function run(capacity: number, ops: any[]): any[] {
  const c = new LRUCache(capacity);
  const out: any[] = [];
  for (const op of ops) {
    if (op[0] === 'put') c.put(op[1], op[2]);
    else if (op[0] === 'get') out.push(c.get(op[1]));
  }
  return out;
}
O(1) per get/put. Both Python's OrderedDict and JS's Map preserve insertion order and let us re-position a key by delete-then-set. The hand-rolled DLL version (in lesson 3) is what some interviewers explicitly request — be ready to write either.

5 — Subsets (bitmask enumeration)

Given an array of distinct integers, return all 2ⁿ subsets. Use bitmask iteration — it's the cleanest solution and demonstrates the technique.

def subsets(nums):
    n = len(nums)
    out = []
    for mask in range(1 << n):
        subset = [nums[i] for i in range(n) if (mask >> i) & 1]
        out.append(subset)
    out.sort()    # for stable test output
    for s in out: s.sort()
    return out
  
function subsets(nums) {
  const n = nums.length;
  const out = [];
  for (let mask = 0; mask < (1 << n); mask++) {
    const sub = [];
    for (let i = 0; i < n; i++) if ((mask >> i) & 1) sub.push(nums[i]);
    out.push(sub);
  }
  out.forEach(s => s.sort((a, b) => a - b));
  out.sort((a, b) => {
    for (let i = 0; i < Math.min(a.length, b.length); i++) {
      if (a[i] !== b[i]) return a[i] - b[i];
    }
    return a.length - b.length;
  });
  return out;
}
  
function subsets(nums: number[]): number[][] {
  const n = nums.length;
  const out: number[][] = [];
  for (let mask = 0; mask < (1 << n); mask++) {
    const sub: number[] = [];
    for (let i = 0; i < n; i++) if ((mask >> i) & 1) sub.push(nums[i]);
    out.push(sub);
  }
  out.forEach(s => s.sort((a, b) => a - b));
  out.sort((a, b) => {
    for (let i = 0; i < Math.min(a.length, b.length); i++) {
      if (a[i] !== b[i]) return a[i] - b[i];
    }
    return a.length - b.length;
  });
  return out;
}
O(n × 2ⁿ) — there are 2ⁿ subsets, each takes O(n) to materialize. The bitmask technique is the cleanest enumeration of subsets — every integer from 0 to 2ⁿ-1 corresponds to exactly one subset. Recursion / backtracking gives the same Big-O but is more code.
