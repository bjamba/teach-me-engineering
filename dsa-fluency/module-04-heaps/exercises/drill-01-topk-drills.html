1 — Top K frequent elements

import heapq
from collections import Counter

def top_k_frequent(nums, k):
    counts = Counter(nums)
    # min-heap of (frequency, value), capped at size k
    heap = []
    for val, freq in counts.items():
        if len(heap) < k:
            heapq.heappush(heap, (freq, val))
        elif freq > heap[0][0]:
            heapq.heapreplace(heap, (freq, val))
    return sorted([v for _, v in heap])
O(n log k) time, O(n + k) space. A heap of size k keeps the constant cheap. The Counter pass is O(n); each heap update is O(log k). Sort at the end is O(k log k) — negligible for k ≪ n. The bucket-sort approach (O(n) total) is also acceptable when frequencies are bounded by n.

2 — Merge K sorted lists

import heapq

def merge_k(lists):
    out = []
    heap = []
    # seed: first element of each non-empty list
    for li, lst in enumerate(lists):
        if lst:
            heapq.heappush(heap, (lst[0], li, 0))
    while heap:
        val, li, ei = heapq.heappop(heap)
        out.append(val)
        if ei + 1 < len(lists[li]):
            heapq.heappush(heap, (lists[li][ei + 1], li, ei + 1))
    return out
O(N log k) time where N = total elements, k = number of lists. Each pop yields one output and pushes at most one replacement, so the heap stays at size ≤ k. Tuples include the list index (li) as a tie-breaker so heapq doesn't try to compare value-equal items.

3 — Median from data stream

import heapq

class MedianFinder:
    def __init__(self):
        self.lo = []   # max-heap (push -x)
        self.hi = []   # min-heap

    def add(self, x):
        # default: push to lo
        heapq.heappush(self.lo, -x)
        # ensure lo's top <= hi's top
        if self.hi and -self.lo[0] > self.hi[0]:
            heapq.heappush(self.hi, -heapq.heappop(self.lo))
        # rebalance sizes (lo can be at most 1 larger than hi)
        if len(self.lo) > len(self.hi) + 1:
            heapq.heappush(self.hi, -heapq.heappop(self.lo))
        elif len(self.hi) > len(self.lo):
            heapq.heappush(self.lo, -heapq.heappop(self.hi))

    def median(self):
        if len(self.lo) > len(self.hi):
            return -self.lo[0]
        return (-self.lo[0] + self.hi[0]) / 2

def run(ops):
    mf = MedianFinder()
    out = []
    for op in ops:
        if op[0] == 'add': mf.add(op[1])
        elif op[0] == 'median': out.append(mf.median())
    return out
O(log n) per add, O(1) per median. The two-heap trick: max-heap of the lower half, min-heap of the upper half. Keep them balanced (sizes differ by at most 1) and verify lo's max ≤ hi's min after each add. The median is either the larger heap's top (odd count) or the average of both tops (even count).
